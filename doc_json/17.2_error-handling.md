## Basic Error Handling[​](#basic-error-handling "Direct link to Basic Error Handling")

All API errors extend from `HttpError` and can be caught generically:

```typescript
import { HttpError } from '@trimble-agentic-external-npm-local/agentic-platform-sdk-ts';

try {
  const agent = await sdk.agents.get('agent-123');
} catch (error) {
  if (error instanceof HttpError) {
    console.error('API Error:', error.errorPayload.title);
    console.error('Details:', error.errorPayload.detail);
    console.error('Status:', error.errorPayload.status);
    console.error('Type:', error.errorPayload.type);
  } else {
    console.error('Unexpected error:', error);
  }
}

```

## Catching Specific Error Types[​](#catching-specific-error-types "Direct link to Catching Specific Error Types")

Use `instanceof` checks to handle specific error scenarios:

```typescript
import {
  ResourceNotFoundError,
  UnauthorizedError,
  PermissionDeniedError,
  InvalidPayloadError,
} from '@trimble-agentic-external-npm-local/agentic-platform-sdk-ts';

async function getAgent(agentId: string) {
  try {
    return await sdk.agents.get(agentId);
  } catch (error) {
    if (error instanceof ResourceNotFoundError) {
      console.log('Agent does not exist');
      return null;
    } else if (error instanceof UnauthorizedError) {
      console.log('Please log in to continue');
      redirectToLogin();
      throw error;
    } else if (error instanceof PermissionDeniedError) {
      console.log('You do not have access to this agent');
      throw error;
    } else {
      throw error;
    }
  }
}

```

### Creating an Agent with Validation[​](#creating-an-agent-with-validation "Direct link to Creating an Agent with Validation")

```typescript
import {
  InvalidPayloadError,
  InvalidModelError,
  BadAgentConfigurationError,
  ConflictError,
} from '@trimble-agentic-external-npm-local/agentic-platform-sdk-ts';

async function createAgent(config: CreateAgentConfigRequest) {
  try {
    const agent = await sdk.agents.create(config);
    return { success: true, agent };
  } catch (error) {
    if (error instanceof InvalidPayloadError) {
      // Handle validation errors
      return {
        success: false,
        error: 'Invalid configuration',
        details: error.errorPayload.detail,
      };
    } else if (error instanceof InvalidModelError) {
      return {
        success: false,
        error: 'Invalid model specified',
        details: 'Please select from available models',
      };
    } else if (error instanceof BadAgentConfigurationError) {
      return {
        success: false,
        error: 'Configuration validation failed',
        details: error.errorPayload.detail,
      };
    } else if (error instanceof ConflictError) {
      return {
        success: false,
        error: 'Agent with this configuration already exists',
      };
    } else {
      throw error;
    }
  }
}

```

      throw new Error('You have exceeded your agent run quota');

## Handling Rate Limits[​](#handling-rate-limits "Direct link to Handling Rate Limits")

Implement exponential backoff for rate limit errors:

```typescript
import {
  RateLimitError,
  RunQuotaExceededError,
  TokenQuotaExceededError,
} from '@trimble-agentic-external-npm-local/agentic-platform-sdk-ts';

async function callWithRetry<T>(
  fn: () => Promise<T>,
  maxRetries: number = 3
): Promise<T> {
  let lastError: Error | undefined;

  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error) {
      if (error instanceof RateLimitError) {
        const delay = Math.pow(2, i) * 1000; // Exponential backoff
        console.log(`Rate limited, waiting ${delay}ms before retry...`);
        await new Promise(resolve => setTimeout(resolve, delay));
        lastError = error;
        continue;
      } else if (error instanceof RunQuotaExceededError) {
        console.error('Run quota exceeded, cannot retry');
        throw new Error('You have exceeded your agent run quota');
      } else if (error instanceof TokenQuotaExceededError) {
        console.error('Token quota exceeded, cannot retry');
        throw new Error('You have exceeded your token quota');
      } else {
        throw error;
      }
    }
  }

  throw lastError || new Error('Max retries exceeded');
}

// Usage
const agents = await callWithRetry(() => sdk.agents.list({ pageSize: 100 }));

```

### Optimistic Updates with Conflict Resolution[​](#optimistic-updates-with-conflict-resolution "Direct link to Optimistic Updates with Conflict Resolution")

```typescript
import {
  StaleObjectError,
  InvalidVersionError,
} from '@trimble-agentic-external-npm-local/agentic-platform-sdk-ts';

async function updateAgentWithConflictResolution(
  agentId: string,
  updates: UpdateAgentConfigRequest,
  maxRetries: number = 3
) {
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      // Get latest version
      const agent = await sdk.agents.get(agentId);

      // Apply updates
      const updated = await sdk.agents.update(agentId, updates);

      return updated;
    } catch (error) {
      if (
        error instanceof StaleObjectError ||
        error instanceof InvalidVersionError
      ) {
        console.log(`Conflict detected, retrying... (attempt ${attempt + 1})`);
        // Retry with latest version
        continue;
      } else {
        throw error;
      }
    }
  }

  throw new Error('Failed to update after conflict resolution retries');
}

```

### Circuit Breaker Pattern[​](#circuit-breaker-pattern "Direct link to Circuit Breaker Pattern")

```typescript
import {
  InternalServerError,
  ServiceUnavailableError,
  BadGatewayError,
} from '@trimble-agentic-external-npm-local/agentic-platform-sdk-ts';

class CircuitBreaker {
  private failureCount = 0;
  private lastFailureTime: number | null = null;
  private readonly threshold = 5;
  private readonly resetTimeoutMs = 60000; // 1 minute

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    // Check if circuit is open
    if (this.isOpen()) {
      throw new Error('Circuit breaker is open, service is unavailable');
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      if (
        error instanceof InternalServerError ||
        error instanceof ServiceUnavailableError ||
        error instanceof BadGatewayError
      ) {
        this.onFailure();
      }
      throw error;
    }
  }

  private isOpen(): boolean {
    if (this.failureCount < this.threshold) {
      return false;
    }

    // Check if we should reset
    if (
      this.lastFailureTime &&
      Date.now() - this.lastFailureTime > this.resetTimeoutMs
    ) {
      this.reset();
      return false;
    }

    return true;
  }

  private onSuccess() {
    this.failureCount = 0;
    this.lastFailureTime = null;
  }

  private onFailure() {
    this.failureCount++;
    this.lastFailureTime = Date.now();
  }

  private reset() {
    this.failureCount = 0;
    this.lastFailureTime = null;
  }
}

// Usage
const breaker = new CircuitBreaker();
const agent = await breaker.execute(() => sdk.agents.get('agent-123'));

```

## Best Practices[​](#best-practices "Direct link to Best Practices")

1. **Always catch specific error types first** - Check for specific error classes before falling back to generic `HttpError`

2. **Provide meaningful error messages** - Transform technical errors into user-friendly messages

3. **Log errors for monitoring** - Always log unexpected errors for debugging and monitoring

4. **Implement retry logic for transient errors** - Automatically retry for rate limits and server errors

5. **Use circuit breakers for degraded services** - Prevent cascading failures by detecting and handling service degradation

6. **Validate before calling the API** - Reduce API errors by validating inputs client-side first

7. **Handle quota errors gracefully** - Provide clear feedback when users hit quotas

8. **Test error handling paths** - Ensure error handlers work correctly by testing error scenarios
